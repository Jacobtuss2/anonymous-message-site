import csv import os from datetime import datetime import sqlite3 from io import StringIO from flask import Flask, render_template, request, redirect, url_for, session, Response from werkzeug.security import generate_password_hash, check_password_hash app = Flask(__name__) app.secret_key = os.environ.get("SECRET_KEY", "change-this-secret-key") DEFAULT_ADMIN_PASSWORD = os.environ.get("ADMIN_PASSWORD", "change-me-strong-password") app.config["ADMIN_PASSWORD_HASH"] = generate_password_hash( DEFAULT_ADMIN_PASSWORD, method="scrypt", ) DB_NAME = "messages.db" def get_db(): conn = sqlite3.connect(DB_NAME) conn.row_factory = sqlite3.Row return conn def init_db(): conn = get_db() conn.execute( """ CREATE TABLE IF NOT EXISTS messages ( id INTEGER PRIMARY KEY AUTOINCREMENT, text TEXT NOT NULL, created_at TEXT NOT NULL, read INTEGER NOT NULL DEFAULT 0, is_private INTEGER NOT NULL DEFAULT 1 ) """ ) conn.commit() columns = conn.execute("PRAGMA table_info(messages)").fetchall() column_names = {col[1] for col in columns} if "read" not in column_names: conn.execute("ALTER TABLE messages ADD COLUMN read INTEGER NOT NULL DEFAULT 0") if "is_private" not in column_names: conn.execute("ALTER TABLE messages ADD COLUMN is_private INTEGER NOT NULL DEFAULT 1") conn.commit() conn.close() def get_message_count(): conn = get_db() count = conn.execute("SELECT COUNT(*) AS total FROM messages WHERE is_private = 1").fetchone()["total"] conn.close() return count def get_recent_messages(limit=5): conn = get_db() messages = conn.execute( "SELECT * FROM messages WHERE is_private = 1 ORDER BY created_at DESC, id DESC LIMIT ?", (limit,), ).fetchall() conn.close() return messages def normalize_sort(value): return value if value in {"newest", "oldest"} else "newest" def normalize_filter(value): return value if value in {"all", "read", "unread"} else "all" def get_messages_for_admin(search="", sort="newest", read_filter="all"): query = "SELECT * FROM messages WHERE is_private = 1" conditions = [] params = [] if search: conditions.append("text LIKE ?") params.append(f"%{search}%") if read_filter == "read": conditions.append("read = ?") params.append(1) elif read_filter == "unread": conditions.append("read = ?") params.append(0) if conditions: query += " AND " + " AND ".join(conditions) if sort == "oldest": query += " ORDER BY created_at ASC, id ASC" else: query += " ORDER BY created_at DESC, id DESC" conn = get_db() rows = conn.execute(query, params).fetchall() conn.close() return rows def get_status_counts(): conn = get_db() totals = conn.execute( "SELECT COUNT(*) AS total, SUM(CASE WHEN read = 1 THEN 1 ELSE 0 END) AS read_count, SUM(CASE WHEN read = 0 THEN 1 ELSE 0 END) AS unread_count FROM messages WHERE is_private = 1" ).fetchone() conn.close() return { "total": totals["total"] or 0, "read": totals["read_count"] or 0, "unread": totals["unread_count"] or 0, } @app.route("/") def index(): sent = request.args.get("sent") == "1" return render_template( "index.html", message_count=get_message_count(), recent_messages=get_recent_messages(5), sent=sent, ) @app.route("/submit", methods=["POST"]) def submit(): message = request.form.get("message", "").strip() if message: now = datetime.now().strftime("%Y-%m-%d %H:%M:%S") conn = get_db() conn.execute( "INSERT INTO messages (text, created_at, read, is_private) VALUES (?, ?, 0, 1)", (message, now), ) conn.commit() conn.close() return redirect(url_for("index", sent="1")) @app.route("/login", methods=["GET", "POST"]) def login(): if request.method == "POST": password = request.form.get("password", "") if check_password_hash(app.config["ADMIN_PASSWORD_HASH"], password): session["admin"] = True return redirect(url_for("admin")) return render_template("login.html", error="Incorrect password.") return render_template("login.html", error=None) @app.route("/logout", methods=["POST"]) def logout(): session.pop("admin", None) return redirect(url_for("login")) @app.route("/admin") def admin(): if not session.get("admin"): return redirect(url_for("login")) search = request.args.get("search", "").strip() sort = normalize_sort(request.args.get("sort")) read_filter = normalize_filter(request.args.get("filter")) messages = get_messages_for_admin(search, sort, read_filter) status = get_status_counts() return render_template( "admin.html", messages=messages, message_count=status["total"], read_count=status["read"], unread_count=status["unread"], search=search, sort=sort, filter=read_filter, ) @app.route("/toggle-read/", methods=["POST"]) def toggle_read(message_id): if not session.get("admin"): return redirect(url_for("login")) conn = get_db() row = conn.execute("SELECT read FROM messages WHERE id = ? AND is_private = 1", (message_id,)).fetchone() if row is not None: conn.execute( "UPDATE messages SET read = ? WHERE id = ?", (0 if row["read"] else 1, message_id), ) conn.commit() conn.close() return redirect(request.referrer or url_for("admin")) @app.route("/delete/", methods=["POST"]) def delete_message(message_id): if not session.get("admin"): return redirect(url_for("login")) conn = get_db() conn.execute("DELETE FROM messages WHERE id = ? AND is_private = 1", (message_id,)) conn.commit() conn.close() return redirect(request.referrer or url_for("admin")) @app.route("/clear-all", methods=["POST"]) def clear_all_messages(): if not session.get("admin"): return redirect(url_for("login")) conn = get_db() conn.execute("DELETE FROM messages WHERE is_private = 1") conn.commit() conn.close() return redirect(url_for("admin")) @app.route("/export") def export_messages(): if not session.get("admin"): return redirect(url_for("login")) search = request.args.get("search", "").strip() sort = normalize_sort(request.args.get("sort")) read_filter = normalize_filter(request.args.get("filter")) rows = get_messages_for_admin(search, sort, read_filter) output = StringIO() writer = csv.writer(output) writer.writerow(["id", "text", "created_at", "read"]) for row in rows: writer.writerow([row["id"], row["text"], row["created_at"], "read" if row["read"] else "unread"]) csv_data = output.getvalue() response = Response(csv_data, mimetype="text/csv") response.headers["Content-Disposition"] = "attachment; filename=messages.csv" return response if __name__ == "__main__": init_db() app.run(host="0.0.0.0", port=5000, debug=True)
